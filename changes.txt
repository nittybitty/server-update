   PROPOSED CHANGE:

   # NEW unified function
   update_server() {
       local server="$1"
       local is_localhost="${2:-false}"
       local show_realtime="${3:-false}"

       # Pre-checks
       [[ "$is_localhost" == "false" ]] && check_disk_space "$server" || return 1

       if ! get_update_approval "$server" "$is_localhost"; then

       fi

       execute_package_update "$server" "$pkg_manager" "$show_realtime"

       # Handle kernel reboot if needed

2. REPEATED FILE READS - Caching Not Implemented
   Problem: Same temp files read multiple times per server

   Package manager file (read 3 times per server):
   - Line 1023: apply_updates()
   - Line 1235: Phase 2 display
   - Line 1466: Phase 4 localhost

   Status file (read 4 times per server):
   - Line 731: draw_dashboard()
   - Line 1211: Phase 2 loop
   - Line 1399: Phase 3.5 results
   - Line 1442: Phase 4 check

   Impact: On 100 servers = 700+ unnecessary file reads with flock overhead
   Solution: Cache reads in associative arrays
   Performance gain: 30-40% fewer file operations

   PROPOSED CHANGE:
   ```bash
   # Add at top after global arrays
   declare -A PKG_MANAGER_CACHE
   declare -A STATUS_CACHE
   declare -A OS_INFO_CACHE
   declare -A KERNEL_INFO_CACHE

   # NEW caching function
   get_pkg_manager() {
       local server="$1"
       if [[ -z "${PKG_MANAGER_CACHE[$server]}" ]]; then
           PKG_MANAGER_CACHE[$server]=$(safe_read_file "$TEMP_DIR/${server}.pkg_manager" "unknown")
       fi
       echo "${PKG_MANAGER_CACHE[$server]}"
   }

   get_status() {
       local server="$1"
       local default="${2:-Unknown}"
       STATUS_CACHE[$server]=$(safe_read_file "$TEMP_DIR/${server}.status" "$default")
       echo "${STATUS_CACHE[$server]}"
   }

   # Replace all instances of:
   # pkg_manager=$(safe_read_file "$TEMP_DIR/${server}.pkg_manager" "unknown")
   # WITH:
   # pkg_manager=$(get_pkg_manager "$server")
   ```

3. INEFFICIENT PACKAGE OUTPUT PARSING - Subshells in Loops
   Location: Lines 1240-1273
   Problem: Spawns subshell for EVERY package line

   Current code:
   ```bash
   grep -v "^Listing\.\.\.$" "$output_file" | while read -r line; do
       if has_kernel_package "$line" "apt"; then
           echo -e "${RED}${BOLD}$line${NC}"
       else
           echo "$line"
       fi
   done
   ```

   Issue: 200 packages = 200+ subshell spawns + 200+ has_kernel_package() calls
   Impact: 50-70% slower for large package lists

   PROPOSED CHANGE - Option 1 (awk, best performance):
   ```bash
   awk -v red="$RED" -v bold="$BOLD" -v nc="$NC" '
       /^Listing\.\.\.$/ { next }
       /linux-image|linux-headers|linux-modules/ {
           printf "%s%s%s%s\n", red, bold, $0, nc
           next
       }
       { print }
   ' "$output_file"
   ```

   PROPOSED CHANGE - Option 2 (two-pass grep, good balance):
   ```bash
   # Identify kernel packages once
   kernel_lines=$(grep -iE "(linux-image|linux-headers|linux-modules)" "$output_file")

   # Color output in one pass
   grep -v "^Listing\.\.\.$" "$output_file" | while IFS= read -r line; do
       if grep -qFx "$line" <<< "$kernel_lines"; then
           echo -e "${RED}${BOLD}$line${NC}"
       else
           echo "$line"
       fi
   done
   ```

4. INEFFICIENT SSH CALLS - Multiple Calls for Related Data
   Location: Lines 532-541 (gather_system_info)
   Problem: Two separate SSH calls for OS and kernel info

   Current code:
   ```bash
   # Call 1: Get OS info
   os_info=$($ssh_cmd "$server" "grep '^PRETTY_NAME=' /etc/os-release | cut -d'\"' -f2")
   # Call 2: Get kernel
   kernel_ver=$($ssh_cmd "$server" "uname -r")
   ```

   Impact: Even with ControlMaster, this is 2x the latency

   PROPOSED CHANGE:
   ```bash
   # Single SSH call returning both values
   read -r os_info kernel_ver <<< "$($ssh_cmd "$server" '
       grep "^PRETTY_NAME=" /etc/os-release 2>/dev/null | cut -d"\"" -f2
       uname -r 2>/dev/null
   ' | tr '\n' ' ')"
   ```

   Location: Lines 631-635 (check_disk_space)
   Problem: Two separate SSH calls for / and /boot

   PROPOSED CHANGE:
   ```bash
   # Single SSH call for both partitions
   $ssh_cmd "$server" '
       df -BG / 2>/dev/null | tail -1 | awk "{print \$4}" | sed "s/G//"
       df -BM /boot 2>/dev/null | tail -1 | awk "{print \$4}" | sed "s/M//"
   ' | {
       read root_avail
       read boot_avail
   }
   ```

   Performance gain: 50% fewer SSH commands in these functions

5. LONG UNMAINTAINABLE FUNCTIONS - Violating Single Responsibility
   Problem: Functions doing too many things, hard to test/debug

   Offenders:
   a) apply_updates() - 117 lines (1017-1134)
      Does: disk check, update execution, kernel detection, reboot, verification

   b) verify_reboot() - 87 lines (929-1016)
      Does: wait for down, wait for up, status updates, logging

   c) check_server_updates() - 130 lines (799-929)
      Does: SSH test, pkg detection, update check, parsing, counting

   PROPOSED CHANGE for apply_updates():
   ```bash
   apply_updates() {
       local server="$1"

       # Step 1: Pre-flight checks
       verify_update_prerequisites "$server" || return 1

       # Step 2: Execute the update
       local update_output
       update_output=$(execute_update_command "$server") || return 1

       # Step 3: Post-update actions
       handle_post_update_actions "$server" "$update_output"
   }

   verify_update_prerequisites() {
       local server="$1"
       check_disk_space "$server" || return 1
       update_status "$server" "Prerequisites passed"
       return 0
   }

   execute_update_command() {
       local server="$1"
       local pkg_manager=$(get_pkg_manager "$server")
       local update_cmd=$(get_update_command "$pkg_manager")

       update_status "$server" "Applying updates (via $pkg_manager)..."
       execute_remote_command "$server" "$update_cmd" "$DNF_TIMEOUT"
   }

   handle_post_update_actions() {
       local server="$1"
       local output="$2"

       if kernel_was_updated "$server" "$output"; then
           safe_write_file "$TEMP_DIR/${server}.update_type" "kernel_update"
           initiate_server_reboot "$server"
       else
           safe_write_file "$TEMP_DIR/${server}.update_type" "no_reboot"
           update_status "$server" "✓ Complete - No reboot needed"
       fi
   }
   ```

   Benefit: Each function <50 lines, testable, debuggable, maintainable
